{{- /* Shared callout component for both notice shortcode and blockquote render hook */ -}} {{- /* Parameters: - style: The style class (note, tip, warning, etc., or "_unset" for no icon) - title: The title text - content: The content to display (template.HTML) - open: Whether the callout should be open by default (optional, default true) - icon: Custom icon name (optional, defaults to style-based icon) - titleOnly: If true, renders a title-only callout (optional, default false) -*/ -}} {{- $style := .style | default "note" -}} {{- $title := .title -}} {{- $content := .content -}} {{- $open := .open | default true -}} {{- $icon := .icon -}} {{- $titleOnly := .titleOnly | default false -}} {{- /* Determine which icon to use */ -}} {{- $iconName := "" -}} {{- if $icon -}} {{- $iconName = $icon -}} {{- else if (ne $style "_unset") -}} {{- $iconName = (printf "notice/%s" $style) -}} {{- end -}} {{- if $titleOnly -}}
{{- with $iconName -}} {{ partial "helper/icon" . }} {{- end -}} {{ $title }}
{{- else -}}
{{- with $iconName -}} {{ partial "helper/icon" . }} {{- end -}} {{ $title }} {{ partial "helper/icon" "chevron-right" }}
{{ $content }}
{{- end -}}
